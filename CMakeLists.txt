cmake_minimum_required(VERSION 3.20)
project(parking_management LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(image_preprocess src/image/image_preprocess.cpp)
target_include_directories(image_preprocess PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(image_preprocess PUBLIC ${OpenCV_LIBS})
add_library(gemini_recognizer src/image/gemini_recognizer.cpp)
target_include_directories(gemini_recognizer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${JSON_INCLUDE_DIR})
target_link_libraries(gemini_recognizer PUBLIC image_preprocess winhttp)
add_executable(parking src/main.cpp src/menus.cpp src/ordinary_user.cpp src/admin_user.cpp src/vehicle.cpp src/parking_service.cpp src/storage.cpp src/fee_calculator.cpp src/plate_recognition.cpp)
target_include_directories(parking PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(parking PRIVATE PARKING_WITH_GEMINI)
target_link_libraries(parking PRIVATE gemini_recognizer)
if(MINGW)
  target_link_options(parking PRIVATE -static)
endif()
add_executable(test_image_preprocess tests/image/test_image_preprocess.cpp)
target_link_libraries(test_image_preprocess PRIVATE image_preprocess)
enable_testing()
add_test(NAME image_preprocess COMMAND test_image_preprocess)
add_executable(test_gemini_response tests/image/test_gemini_response.cpp)
target_link_libraries(test_gemini_response PRIVATE gemini_recognizer)
add_test(NAME gemini_response COMMAND test_gemini_response)
